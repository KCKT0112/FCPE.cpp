cmake_minimum_required(VERSION 3.18)
project(fcpe VERSION 0.1.0 LANGUAGES C CXX)

option(FCPE_BUILD_CLI "Build the WAV / mel command line tool" ON)
option(FCPE_BUILD_TESTS "Build tests" ON)
option(FCPE_BUILD_BENCHMARK "Build reusable-session benchmark" ON)
option(FCPE_CUDA "Enable ggml CUDA" OFF)
option(FCPE_VULKAN "Enable ggml Vulkan" OFF)
option(FCPE_METAL "Enable ggml Metal" OFF)
option(FCPE_METAL_STRICT_F32 "Preserve F32 operands in ggml Metal (requires embedded shaders)" ON)
option(FCPE_VULKAN_STRICT_F32 "Preserve F32 arithmetic in ggml Vulkan (recommended)" ON)
option(FCPE_VULKAN_TILED_F32 "Enable tuned and legacy tiled Vulkan F32 matmul (requires strict F32)" ON)
option(FCPE_VULKAN_FUSION "Fuse FCPE Vulkan normalization/GLU/activations (requires strict F32)" ON)
set(FCPE_GGML_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ggml" CACHE PATH "Existing ggml source checkout")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" CONFIG_UPPER)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${CMAKE_BINARY_DIR}/bin")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${CMAKE_BINARY_DIR}/bin")
endforeach()

set(_fcpe_owns_ggml OFF)
if(NOT TARGET ggml)
    set(_fcpe_owns_ggml ON)
    # Some restricted Windows shells omit PROCESSOR_ARCHITECTURE from the environment.
    if(MSVC AND NOT CMAKE_SYSTEM_PROCESSOR AND CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
        set(CMAKE_SYSTEM_PROCESSOR AMD64)
    endif()
    set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GGML_CUDA ${FCPE_CUDA} CACHE BOOL "" FORCE)
    set(GGML_VULKAN ${FCPE_VULKAN} CACHE BOOL "" FORCE)
    set(GGML_METAL ${FCPE_METAL} CACHE BOOL "" FORCE)
    if(EXISTS "${FCPE_GGML_SOURCE_DIR}/CMakeLists.txt")
        add_subdirectory("${FCPE_GGML_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/ggml")
    else()
        include(FetchContent)
        FetchContent_Declare(ggml
            URL https://github.com/ggml-org/ggml/archive/refs/tags/v0.19.0.tar.gz
            URL_HASH SHA256=cfb6512adda2853e6500a7c5b23f326987cb4c723e9f8f93c6c5a7e7e4861648)
        FetchContent_MakeAvailable(ggml)
    endif()
endif()

if(FCPE_VULKAN AND FCPE_VULKAN_STRICT_F32)
    if(NOT _fcpe_owns_ggml)
        message(FATAL_ERROR "For strict Vulkan F32, let FCPE configure ggml using FCPE_GGML_SOURCE_DIR")
    endif()
    include(cmake/VulkanPrecision.cmake)
    fcpe_vulkan_precision()
endif()

if(FCPE_METAL AND FCPE_METAL_STRICT_F32)
    if(NOT _fcpe_owns_ggml)
        message(FATAL_ERROR "For strict Metal F32, let FCPE configure ggml using FCPE_GGML_SOURCE_DIR")
    endif()
    include(cmake/MetalPrecision.cmake)
    fcpe_metal_precision()
endif()

add_library(fcpe STATIC src/fcpe.cpp src/audio.cpp)
add_library(fcpe::fcpe ALIAS fcpe)
if(FCPE_METAL AND FCPE_METAL_STRICT_F32)
    target_compile_definitions(fcpe PRIVATE FCPE_METAL_OPTIMIZED)
endif()
target_include_directories(fcpe PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include> PRIVATE third_party)
find_package(Threads REQUIRED)
target_link_libraries(fcpe PUBLIC ggml PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(fcpe PRIVATE /W4 /utf-8)
else()
    target_compile_options(fcpe PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FCPE_BUILD_CLI)
    add_executable(fcpe-cli tools/fcpe-cli.cpp)
    target_link_libraries(fcpe-cli PRIVATE fcpe::fcpe)
endif()
if(FCPE_BUILD_TESTS)
    enable_testing()
    add_executable(fcpe-tests tests/test_fcpe.cpp)
    target_link_libraries(fcpe-tests PRIVATE fcpe::fcpe)
    add_test(NAME fcpe-unit COMMAND fcpe-tests)
    add_executable(fcpe-backend-tests tests/test_backend.cpp)
    target_link_libraries(fcpe-backend-tests PRIVATE ggml)
    add_test(NAME fcpe-backend-cpu COMMAND fcpe-backend-tests CPU)
    add_executable(fcpe-fusion-tests tests/test_fusion.cpp)
    target_link_libraries(fcpe-fusion-tests PRIVATE ggml)
    add_executable(fcpe-metal-ops-tests tests/test_metal_ops.cpp)
    target_link_libraries(fcpe-metal-ops-tests PRIVATE ggml)
    set(FCPE_TEST_MODEL "" CACHE FILEPATH "Optional converted GGUF for model tests")
    set(FCPE_TEST_BACKEND "" CACHE STRING "Optional accelerator device to test, e.g. MTL0 or Vulkan0")
    if(FCPE_TEST_BACKEND)
        add_test(NAME fcpe-backend-accelerator COMMAND fcpe-backend-tests "${FCPE_TEST_BACKEND}")
        if(FCPE_METAL)
            add_test(NAME fcpe-metal-ops COMMAND fcpe-metal-ops-tests "${FCPE_TEST_BACKEND}")
            add_test(NAME fcpe-metal-fusion COMMAND fcpe-fusion-tests "${FCPE_TEST_BACKEND}")
        endif()
    endif()
    if(FCPE_TEST_MODEL)
        add_test(NAME fcpe-model COMMAND fcpe-tests "${FCPE_TEST_MODEL}")
        if(FCPE_TEST_BACKEND)
            add_test(NAME fcpe-model-accelerator COMMAND fcpe-tests "${FCPE_TEST_MODEL}" "${FCPE_TEST_BACKEND}")
        endif()
    endif()
endif()
if(FCPE_BUILD_BENCHMARK)
    add_executable(fcpe-bench tools/fcpe-bench.cpp)
    target_link_libraries(fcpe-bench PRIVATE fcpe::fcpe)
endif()
include(GNUInstallDirs)
install(TARGETS fcpe ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/fcpe DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE NOTICE.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/licenses)
if(FCPE_BUILD_CLI)
    install(TARGETS fcpe-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
